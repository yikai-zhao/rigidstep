cmake_minimum_required(VERSION 3.20)
project(rigidstep VERSION 0.1.0 LANGUAGES CXX)
option(RIGIDSTEP_SANITIZERS "Enable AddressSanitizer and UBSan (GCC/Clang)" OFF)

add_library(rigidstep_lib src/physics.cpp)
target_include_directories(rigidstep_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rigidstep_lib PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(rigidstep_lib PRIVATE /W4 /permissive-)
else()
    target_compile_options(rigidstep_lib PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(RIGIDSTEP_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer option requires GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(rigidstep_lib PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rigidstep_lib PRIVATE -fsanitize=address,undefined)
endif()
add_executable(rigidstep apps/rigidstep.cpp)
target_link_libraries(rigidstep PRIVATE rigidstep_lib)
add_executable(rigidstep_tests tests/test_physics.cpp)
target_link_libraries(rigidstep_tests PRIVATE rigidstep_lib)
enable_testing()
add_test(NAME physics_correctness COMMAND rigidstep_tests)
